name: i18n Checks
on: [push, pull_request]
jobs:
  check:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
      - run: npm ci
      - run: npx @angular-translation-service/cli generate -i src/i18n/en --check
      - run: npx @angular-translation-service/cli validate -i src/i18n --default-lang en
      - run: npx @angular-translation-service/cli check --i18n src/i18n --src src
